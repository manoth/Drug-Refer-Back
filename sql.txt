SELECT 
(SELECT hospitalcode FROM opdconfig LIMIT 1) as hospmain,
o.hn,
o.vn,
p.pname, 
p.fname, 
p.lname,
p.cid,
p.birthday,
p.informaddr,
p.hometel as tel,
p.informtel as telinform,
n.ename as dwh_hospcode,
n.ename as dwh_code,
n.name as dwh_name,
(SELECT GROUP_CONCAT(DISTINCT CONCAT(da.icd10, '|', IFNULL(REPLACE(i.name, ',', ' '), '')) SEPARATOR '$$$') 
FROM ovstdiag da LEFT JOIN icd9cm1 i ON da.icd10=i.code WHERE da.hn=o.hn AND da.vn=o.vn AND SUBSTR(da.icd10,1,1)>0) as icd9,
(SELECT GROUP_CONCAT(DISTINCT CONCAT(da.icd10, '|',IFNULL(REPLACE(i.name, ',', ' '), ''), '|', IFNULL(REPLACE(i.tname, ',', ' '), '')) SEPARATOR '$$$') 
FROM ovstdiag da LEFT JOIN icd101 i ON da.icd10=i.code WHERE da.hn=o.hn AND da.vn=o.vn AND SUBSTR(da.icd10,1,1)<=0) as icd10,
s.cc, s.pe, s.hpi, od.diag_text,
(SELECT c.name FROM clinicmember m INNER JOIN clinic c ON c.clinic=m.clinic WHERE m.hn=o.hn LIMIT 1) as clinicname,
p.drugallergy as drug_allergy,
k.depcode,
k.department,
k.phone_number as dep_phone_number,
o.vstdate,
(SELECT GROUP_CONCAT(DISTINCT CONCAT(d.did,'|',d.icode,'|',d.name,' ',d.strength,'|',oo.qty,'|',d.units,'|',d.unitprice,'|',u.name1,' ',u.name2,' ',IFNULL(u.name3,'')) SEPARATOR '$$$') 
FROM opitemrece oo INNER JOIN drugitems d ON oo.icode=d.icode 
LEFT JOIN drugusage u ON oo.drugusage=u.drugusage
WHERE oo.hn=o.hn AND oo.vn=o.vn
AND d.did IS NOT NULL
AND oo.icode IN('1000959','1600067','1000566','1620079')) as drug_13,
(SELECT GROUP_CONCAT(DISTINCT CONCAT(d.did,'|',d.icode,'|',d.name,' ',d.strength,'|',oo.qty,'|',d.units,'|',d.unitprice,'|',u.name1,' ',u.name2,' ',IFNULL(u.name3,'')) SEPARATOR '$$$')
FROM opitemrece oo INNER JOIN drugitems d ON oo.icode=d.icode 
LEFT JOIN drugusage u ON oo.drugusage=u.drugusage
WHERE oo.hn=o.hn AND oo.vn=o.vn
AND d.did IS NOT NULL
AND oo.icode NOT IN('1000959','1600067','1000566','1620079')) as drug_other,
(SELECT GROUP_CONCAT(DISTINCT CONCAT(IFNULL(li.lab_items_name,''),'|',IFNULL(lo.lab_order_result,''),'|',IFNULL(li.service_price,'')) SEPARATOR '$$$')
FROM lab_head lh
INNER JOIN lab_order lo ON lh.lab_order_number = lo.lab_order_number
INNER JOIN lab_items li ON lo.lab_items_code = li.lab_items_code
WHERE lh.hn=o.hn AND lh.vn=o.vn) as lab
FROM opitemrece o
INNER JOIN patient p ON p.hn=o.hn
INNER JOIN nondrugitems n ON n.icode=o.icode 
LEFT JOIN opdscreen s ON s.vn=o.vn
LEFT JOIN ovst_doctor_diag od ON od.vn=o.vn
LEFT JOIN kskdepartment k ON k.depcode=o.dep_code
WHERE n.name IS NOT NULL /* TEST DATABASE: restore LIKE '19Drug Refer Back%' before production */
AND o.hn IS NOT NULL AND o.vn IS NOT NULL
AND DATE(o.vstdate) BETWEEN CURDATE() - INTERVAL 1 DAY AND CURDATE()
AND o.vn = ?
GROUP BY hospmain, o.hn, o.vn
